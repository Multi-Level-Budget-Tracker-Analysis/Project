Pre-Built DAX Measures Library

Paste directly into Power BI Desktop Model View to calculate business metrics

// -------------------------------------------------------------
// TIER 1: INDIVIDUAL MEASURES
// -------------------------------------------------------------
Total Income = 
CALCULATE(
    SUM('Fact_Individual'[Amount]),
    'Fact_Individual'[IncomeExpense] = "Income"
)

Total Expenses = 
CALCULATE(
    SUM('Fact_Individual'[Amount]),
    'Fact_Individual'[IncomeExpense] = "Expense"
)

Net Savings = [Total Income] - [Total Expenses]

Savings Rate % = 
DIVIDE([Net Savings], [Total Income], 0) * 100

// -------------------------------------------------------------
// TIER 2: ORGANIZATION MEASURES
// -------------------------------------------------------------
Total Sales = SUM('Fact_Organization'[Sales])

Operating Cost = SUM('Fact_Organization'[Cost])

Operating Profit = SUM('Fact_Organization'[Profit])

Operating Profit Margin % = 
DIVIDE([Operating Profit], [Total Sales], 0) * 100

// -------------------------------------------------------------
// TIER 3: STATE & NATION MEASURES
// -------------------------------------------------------------
State Fiscal Deficit % of GSDP = 
DIVIDE(
    SUM('Fact_StateFinance'[FiscalDeficit]),
    SUM('Fact_StateFinance'[GSDP]),
    0
) * 100

FRBM Compliance Flag = 
IF([State Fiscal Deficit % of GSDP] <= 3.0, "Compliant", "Non-Compliant")
